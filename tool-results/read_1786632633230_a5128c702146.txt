  2129→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
  2130→  const { ref, inView } = useInView(0.3);
  2131→  const [count, setCount] = useState(0);
  2132→
  2133→  useEffect(() => {
  2134→    if (!inView) return;
  2135→    let raf: number;
  2136→    const duration = 1200;
  2137→    const start = performance.now();
  2138→
  2139→    const step = (now: number) => {
  2140→      const elapsed = now - start;
  2141→      const progress = Math.min(elapsed / duration, 1);
  2142→      // Ease-out cubic
  2143→      const eased = 1 - Math.pow(1 - progress, 3);
  2144→      setCount(Math.round(eased * target));
  2145→      if (progress < 1) {
  2146→        raf = requestAnimationFrame(step);
  2147→      }
  2148→    };
  2149→    raf = requestAnimationFrame(step);
  2150→    return () => cancelAnimationFrame(raf);
  2151→  }, [inView, target]);
  2152→
  2153→  return (
  2154→    <span ref={ref}>
  2155→      {prefix}{count}{suffix}
  2156→    </span>
  2157→  );
  2158→}

  2202→function ParticleMesh({
  2203→  className = '',
  2204→  particleCount = 40,
  2205→  maxDistance = 150,
  2206→  color = 'oklch(0.72 0.17 162',
  2207→}: {
  2208→  className?: string;
  2209→  particleCount?: number;
  2210→  maxDistance?: number;
  2211→  color?: string;
  2212→}) {
  2213→  const canvasRef = useRef<HTMLCanvasElement>(null);
  2214→  const particlesRef = useRef<Array<{ x: number; y: number; vx: number; vy: number }>>([]);
  2215→  const animRef = useRef<number>(0);
  2216→  const mouseRef = useRef({ x: -1000, y: -1000 });
  2217→
  2218→  const initParticles = useCallback((w: number, h: number) => {
  2219→    particlesRef.current = Array.from({ length: particleCount }, () => ({
  2220→      x: Math.random() * w,
  2221→      y: Math.random() * h,
  2222→      vx: (Math.random() - 0.5) * 0.3,
  2223→      vy: (Math.random() - 0.5) * 0.3,
  2224→    }));
  2225→  }, [particleCount]);
  2226→
  2227→  const draw = useCallback((ctx: CanvasRenderingContext2D, w: number, h: number) => {
  2228→    ctx.clearRect(0, 0, w, h);
  2229→    const particles = particlesRef.current;
  2230→    const mx = mouseRef.current.x;
  2231→    const my = mouseRef.current.y;
  2232→    const attractRadius = 200;
  2233→
  2234→    for (let i = 0; i < particles.length; i++) {
  2235→      const p = particles[i];
  2236→
  2237→      // S2: Attract particles toward mouse
  2238→      if (mx > 0 && my > 0) {
  2239→        const dx = mx - p.x;
  2240→        const dy = my - p.y;
  2241→        const dist = Math.sqrt(dx * dx + dy * dy);
  2242→        if (dist < attractRadius && dist > 0) {
  2243→          const force = 0.015 * (1 - dist / attractRadius);
  2244→          p.vx += (dx / dist) * force;
  2245→          p.vy += (dy / dist) * force;
  2246→        }
  2247→      }
  2248→
  2249→      // Dampen velocity
  2250→      p.vx *= 0.99;
  2251→      p.vy *= 0.99;
  2252→
  2253→      p.x += p.vx;
  2254→      p.y += p.vy;
  2255→      if (p.x < 0 || p.x > w) p.vx *= -1;
  2256→      if (p.y < 0 || p.y > h) p.vy *= -1;
  2257→      // Clamp position
  2258→      p.x = Math.max(0, Math.min(w, p.x));
  2259→      p.y = Math.max(0, Math.min(h, p.y));
  2260→
  2261→      // Draw particle
  2262→      ctx.beginPath();
  2263→      ctx.arc(p.x, p.y, 1.5, 0, Math.PI * 2);
  2264→      ctx.fillStyle = `${color} / 0.15)`;
  2265→      ctx.fill();
  2266→
  2267→      // Draw lines to nearby particles
  2268→      for (let j = i + 1; j < particles.length; j++) {
  2269→        const p2 = particles[j];
  2270→        const ddx = p.x - p2.x;
  2271→        const ddy = p.y - p2.y;
  2272→        const dist = Math.sqrt(ddx * ddx + ddy * ddy);
  2273→        if (dist < maxDistance) {
  2274→          ctx.beginPath();
  2275→          ctx.moveTo(p.x, p.y);
  2276→          ctx.lineTo(p2.x, p2.y);
  2277→          ctx.strokeStyle = `${color} / 0.06)`;
  2278→          ctx.lineWidth = 0.5;
  2279→          ctx.stroke();
  2280→        }
  2281→      }
  2282→    }
  2283→
  2284→    // S2: Draw mouse glow
  2285→    if (mx > 0 && my > 0) {
  2286→      const gradient = ctx.createRadialGradient(mx, my, 0, mx, my, 200);
  2287→      gradient.addColorStop(0, `${color} / 0.08)`);
  2288→      gradient.addColorStop(1, `${color} / 0)`);
  2289→      ctx.beginPath();
  2290→      ctx.arc(mx, my, 200, 0, Math.PI * 2);
  2291→      ctx.fillStyle = gradient;
  2292→      ctx.fill();
  2293→    }
  2294→  }, [maxDistance, color]);
  2295→
  2296→  useEffect(() => {
  2297→    const canvas = canvasRef.current;
  2298→    if (!canvas) return;
  2299→    const ctx = canvas.getContext('2d');
  2300→    if (!ctx) return;
  2301→
  2302→    const w = window.innerWidth;
  2303→    const h = window.innerHeight;
  2304→    canvas.width = w;
  2305→    canvas.height = h;
  2306→    initParticles(w, h);
  2307→
  2308→    const animate = () => {
  2309→      draw(ctx, w, h);
  2310→      animRef.current = requestAnimationFrame(animate);
  2311→    };
  2312→    animate();
  2313→
  2314→    return () => cancelAnimationFrame(animRef.current);
  2315→  }, [initParticles, draw]);
  2316→
  2317→  // Handle resize
  2318→  useEffect(() => {
  2319→    const handleResize = () => {
  2320→      const canvas = canvasRef.current;
  2321→      if (!canvas) return;
  2322→      const w = window.innerWidth;
  2323→      const h = window.innerHeight;
  2324→      canvas.width = w;
  2325→      canvas.height = h;
  2326→      initParticles(w, h);
  2327→    };
  2328→
  2329→    window.addEventListener('resize', handleResize);
  2330→    return () => window.removeEventListener('resize', handleResize);
  2331→  }, [initParticles]);
  2332→
  2333→  // S2: Track mouse position via window events
  2334→  useEffect(() => {
  2335→    const handleMouseMove = (e: MouseEvent) => {
  2336→      const canvas = canvasRef.current;
  2337→      if (!canvas) return;
  2338→      const rect = canvas.getBoundingClientRect();
  2339→      mouseRef.current = {
  2340→        x: e.clientX - rect.left,
  2341→        y: e.clientY - rect.top,
  2342→      };
  2343→    };
  2344→    const handleMouseLeave = () => {
  2345→      mouseRef.current = { x: -1000, y: -1000 };
  2346→    };
  2347→    window.addEventListener('mousemove', handleMouseMove);
  2348→    window.addEventListener('mouseleave', handleMouseLeave);
  2349→    return () => {
  2350→      window.removeEventListener('mousemove', handleMouseMove);
  2351→      window.removeEventListener('mouseleave', handleMouseLeave);
  2352→    };
  2353→  }, []);
  2354→
  2355→  return <canvas ref={canvasRef} className={className} />;
  2356→}

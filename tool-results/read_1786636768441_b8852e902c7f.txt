   142→function useInView(threshold = 0.15) {
   143→  const ref = useRef<HTMLDivElement>(null);
   144→  const [inView, setInView] = useState(false);
   145→  useEffect(() => {
   146→    const el = ref.current;
   147→    if (!el) return;
   148→    const obs = new IntersectionObserver(
   149→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
   150→      { threshold }
   151→    );
   152→    obs.observe(el);
   153→    return () => obs.disconnect();
   154→  }, [threshold]);
   155→  return { ref, inView };
   156→}
   157→
   158→/* S4: Section reveal hook */
   159→function useSectionReveal() {
   160→  const ref = useRef<HTMLElement>(null);
   161→  useEffect(() => {
   162→    const el = ref.current;
   163→    if (!el) return;
   164→    const obs = new IntersectionObserver(
   165→      ([e]) => { if (e.isIntersecting) { el.classList.add('revealed'); obs.disconnect(); } },
   166→      { threshold: 0.1 }
   167→    );
   168→    obs.observe(el);
   169→    return () => obs.disconnect();
   170→  }, []);
   171→  return ref;
   172→}

  3104→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {

  3110→    let raf: number;

  3112→    const start = performance.now();
  3113→
  3114→    const step = (now: number) => {
  3115→      const elapsed = now - start;
  3116→      const progress = Math.min(elapsed / duration, 1);
  3117→      // Ease-out cubic
  3118→      const eased = 1 - Math.pow(1 - progress, 3);
  3119→      setCount(Math.round(eased * target));
  3120→      if (progress < 1) {

  3125→    return () => cancelAnimationFrame(raf);
  3126→  }, [inView, target]);
  3127→
  3128→  return (
  3129→    <span ref={ref}>
  3130→      {prefix}{count}{suffix}

  3177→function ParticleMesh({
  3178→  className = '',
  3179→  particleCount = 40,
  3180→  maxDistance = 150,
  3181→  color = 'oklch(0.72 0.17 162',
  3182→}: {
  3183→  className?: string;
  3184→  particleCount?: number;
  3185→  maxDistance?: number;
  3186→  color?: string;
  3187→}) {
  3188→  const canvasRef = useRef<HTMLCanvasElement>(null);
  3189→  const particlesRef = useRef<Array<{ x: number; y: number; vx: number; vy: number }>>([]);
  3190→  const animRef = useRef<number>(0);
  3191→  const mouseRef = useRef({ x: -1000, y: -1000 });
  3192→
  3193→  const initParticles = useCallback((w: number, h: number) => {
  3194→    particlesRef.current = Array.from({ length: particleCount }, () => ({
  3195→      x: Math.random() * w,
  3196→      y: Math.random() * h,
  3197→      vx: (Math.random() - 0.5) * 0.3,
  3198→      vy: (Math.random() - 0.5) * 0.3,
  3199→    }));
  3200→  }, [particleCount]);
  3201→
  3202→  const draw = useCallback((ctx: CanvasRenderingContext2D, w: number, h: number) => {
  3203→    ctx.clearRect(0, 0, w, h);
  3204→    const particles = particlesRef.current;
  3205→    const mx = mouseRef.current.x;
  3206→    const my = mouseRef.current.y;
  3207→    const attractRadius = 200;
  3208→
  3209→    for (let i = 0; i < particles.length; i++) {
  3210→      const p = particles[i];

  3213→      if (mx > 0 && my > 0) {
  3214→        const dx = mx - p.x;

  3217→        if (dist < attractRadius && dist > 0) {
  3218→          const force = 0.015 * (1 - dist / attractRadius);
  3219→          p.vx += (dx / dist) * force;
  3220→          p.vy += (dy / dist) * force;
  3221→        }
  3222→      }
  3223→
  3224→      // Dampen velocity
  3225→      p.vx *= 0.99;
  3226→      p.vy *= 0.99;

  3229→      p.y += p.vy;
  3230→      if (p.x < 0 || p.x > w) p.vx *= -1;
  3231→      if (p.y < 0 || p.y > h) p.vy *= -1;

  3233→      p.x = Math.max(0, Math.min(w, p.x));
  3234→      p.y = Math.max(0, Math.min(h, p.y));
  3235→

  3246→        const ddy = p.y - p2.y;

  3250→          ctx.moveTo(p.x, p.y);
  3251→          ctx.lineTo(p2.x, p2.y);

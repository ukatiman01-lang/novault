    91→function useInView(threshold = 0.15) {
    92→  const ref = useRef<HTMLDivElement>(null);
    93→  const [inView, setInView] = useState(false);
    94→  useEffect(() => {
    95→    const el = ref.current;
    96→    if (!el) return;
    97→    const obs = new IntersectionObserver(
    98→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    99→      { threshold }
   100→    );
   101→    obs.observe(el);
   102→    return () => obs.disconnect();
   103→  }, [threshold]);
   104→  return { ref, inView };
   105→}

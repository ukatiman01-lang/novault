    57→function useInView(threshold = 0.15) {
    58→  const ref = useRef<HTMLDivElement>(null);
    59→  const [inView, setInView] = useState(false);
    60→  useEffect(() => {
    61→    const el = ref.current;
    62→    if (!el) return;
    63→    const obs = new IntersectionObserver(
    64→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    65→      { threshold }
    66→    );
    67→    obs.observe(el);
    68→    return () => obs.disconnect();
    69→  }, [threshold]);
    70→  return { ref, inView };
    71→}

  1023→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
  1024→  const { ref, inView } = useInView(0.3);
  1025→  const [count, setCount] = useState(0);
  1026→
  1027→  useEffect(() => {
  1028→    if (!inView) return;
  1029→    let raf: number;
  1030→    const duration = 1200;
  1031→    const start = performance.now();
  1032→
  1033→    const step = (now: number) => {
  1034→      const elapsed = now - start;
  1035→      const progress = Math.min(elapsed / duration, 1);
  1036→      // Ease-out cubic
  1037→      const eased = 1 - Math.pow(1 - progress, 3);
  1038→      setCount(Math.round(eased * target));
  1039→      if (progress < 1) {
  1040→        raf = requestAnimationFrame(step);
  1041→      }
  1042→    };
  1043→    raf = requestAnimationFrame(step);
  1044→    return () => cancelAnimationFrame(raf);
  1045→  }, [inView, target]);
  1046→
  1047→  return (
  1048→    <span ref={ref}>
  1049→      {prefix}{count}{suffix}
  1050→    </span>
  1051→  );
  1052→}

    61→function useInView(threshold = 0.15) {
    62→  const ref = useRef<HTMLDivElement>(null);
    63→  const [inView, setInView] = useState(false);
    64→  useEffect(() => {
    65→    const el = ref.current;
    66→    if (!el) return;
    67→    const obs = new IntersectionObserver(
    68→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    69→      { threshold }
    70→    );
    71→    obs.observe(el);
    72→    return () => obs.disconnect();
    73→  }, [threshold]);
    74→  return { ref, inView };
    75→}

  1436→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
  1437→  const { ref, inView } = useInView(0.3);
  1438→  const [count, setCount] = useState(0);
  1439→
  1440→  useEffect(() => {
  1441→    if (!inView) return;
  1442→    let raf: number;
  1443→    const duration = 1200;
  1444→    const start = performance.now();
  1445→
  1446→    const step = (now: number) => {
  1447→      const elapsed = now - start;
  1448→      const progress = Math.min(elapsed / duration, 1);
  1449→      // Ease-out cubic
  1450→      const eased = 1 - Math.pow(1 - progress, 3);
  1451→      setCount(Math.round(eased * target));
  1452→      if (progress < 1) {
  1453→        raf = requestAnimationFrame(step);
  1454→      }
  1455→    };
  1456→    raf = requestAnimationFrame(step);
  1457→    return () => cancelAnimationFrame(raf);
  1458→  }, [inView, target]);
  1459→
  1460→  return (
  1461→    <span ref={ref}>
  1462→      {prefix}{count}{suffix}
  1463→    </span>
  1464→  );
  1465→}

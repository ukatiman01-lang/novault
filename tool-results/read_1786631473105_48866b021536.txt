    71→function useInView(threshold = 0.15) {
    72→  const ref = useRef<HTMLDivElement>(null);
    73→  const [inView, setInView] = useState(false);
    74→  useEffect(() => {
    75→    const el = ref.current;
    76→    if (!el) return;
    77→    const obs = new IntersectionObserver(
    78→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    79→      { threshold }
    80→    );
    81→    obs.observe(el);
    82→    return () => obs.disconnect();
    83→  }, [threshold]);
    84→  return { ref, inView };
    85→}

   360→  // Navbar scroll detection + scroll progress + back-to-top

   365→      setShowBackToTop(sy > 400);
   366→      const docH = document.body.scrollHeight - window.innerHeight;
   367→      setScrollProgress(docH > 0 ? (sy / docH) * 100 : 0);
   368→    };
   369→    window.addEventListener('scroll', onScroll, { passive: true });
   370→    return () => window.removeEventListener('scroll', onScroll);
   371→  }, []);
   372→
   373→  // Fetch live metrics
   374→  useEffect(() => {
   375→    fetch('/api/metrics')
   376→      .then(r => r.json())
   377→      .then(setLiveMetrics)
   378→      .catch(() => {}); // Silently fail, use defaults
   379→  }, []);
   380→
   381→  // Fetch waitlist count
   382→  useEffect(() => {
   383→    fetch('/api/waitlist')
   384→      .then(r => r.json())
   385→      .then(d => setWaitlistCount(d.count ?? null))
   386→      .catch(() => {});
   387→  }, []);

   395→
   396→  const handleCopyCode = useCallback(() => {

   401→
   402→  const handleContact = async () => {
   403→    if (!contactForm.name || !contactForm.email || !contactForm.message) {
   404→      toast.error('Please fill in all required fields.');
   405→      return;
   406→    }

   409→      const res = await fetch('/api/contact', {
   410→        method: 'POST',
   411→        headers: { 'Content-Type': 'application/json' },
   412→        body: JSON.stringify(contactForm),
   413→      });

   416→        toast.success('Message sent!', { description: "We'll get back to you within 24 hours." });

   419→        toast.error(data.error || 'Something went wrong.');
   420→      }
   421→    } catch {
   422→      toast.error('Failed to send message. Please try again.');
   423→    }
   424→    setContactSending(false);
   425→  };
   426→
   427→  const handleWaitlist = useCallback(async () => {
   428→    if (!waitlistEmail || !waitlistEmail.includes('@')) {
   429→      toast.error('Please enter a valid email address.');
   430→      return;
   431→    }
   432→    try {
   433→      const res = await fetch('/api/waitlist', {
   434→        method: 'POST',
   435→        headers: { 'Content-Type': 'application/json' },
   436→        body: JSON.stringify({ email: waitlistEmail }),
   437→      });
   438→      const data = await res.json();
   439→      if (data.success) {
   440→        setWaitlistCount(data.count);
   441→        toast.success("You're on the list!", { description: `You're #${data.count} on the list!` });

  1701→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
  1702→  const { ref, inView } = useInView(0.3);
  1703→  const [count, setCount] = useState(0);
  1704→
  1705→  useEffect(() => {
  1706→    if (!inView) return;
  1707→    let raf: number;
  1708→    const duration = 1200;
  1709→    const start = performance.now();
  1710→
  1711→    const step = (now: number) => {
  1712→      const elapsed = now - start;
  1713→      const progress = Math.min(elapsed / duration, 1);
  1714→      // Ease-out cubic
  1715→      const eased = 1 - Math.pow(1 - progress, 3);
  1716→      setCount(Math.round(eased * target));
  1717→      if (progress < 1) {
  1718→        raf = requestAnimationFrame(step);
  1719→      }
  1720→    };
  1721→    raf = requestAnimationFrame(step);
  1722→    return () => cancelAnimationFrame(raf);
  1723→  }, [inView, target]);
  1724→
  1725→  return (
  1726→    <span ref={ref}>
  1727→      {prefix}{count}{suffix}
  1728→    </span>
  1729→  );
  1730→}

  1767→function ParticleMesh({
  1768→  className = '',
  1769→  particleCount = 40,
  1770→  maxDistance = 150,
  1771→  color = 'oklch(0.72 0.17 162',
  1772→}: {
  1773→  className?: string;
  1774→  particleCount?: number;
  1775→  maxDistance?: number;
  1776→  color?: string;
  1777→}) {
  1778→  const canvasRef = useRef<HTMLCanvasElement>(null);
  1779→  const particlesRef = useRef<Array<{ x: number; y: number; vx: number; vy: number }>>([]);
  1780→  const animRef = useRef<number>(0);
  1781→
  1782→  const initParticles = useCallback((w: number, h: number) => {
  1783→    particlesRef.current = Array.from({ length: particleCount }, () => ({
  1784→      x: Math.random() * w,
  1785→      y: Math.random() * h,
  1786→      vx: (Math.random() - 0.5) * 0.3,
  1787→      vy: (Math.random() - 0.5) * 0.3,
  1788→    }));
  1789→  }, [particleCount]);
  1790→
  1791→  const draw = useCallback((ctx: CanvasRenderingContext2D, w: number, h: number) => {
  1792→    ctx.clearRect(0, 0, w, h);
  1793→    const particles = particlesRef.current;
  1794→
  1795→    for (let i = 0; i < particles.length; i++) {
  1796→      const p = particles[i];
  1797→      p.x += p.vx;
  1798→      p.y += p.vy;
  1799→      if (p.x < 0 || p.x > w) p.vx *= -1;
  1800→      if (p.y < 0 || p.y > h) p.vy *= -1;
  1801→
  1802→      // Draw particle
  1803→      ctx.beginPath();
  1804→      ctx.arc(p.x, p.y, 1.5, 0, Math.PI * 2);
  1805→      ctx.fillStyle = `${color} / 0.15)`;
  1806→      ctx.fill();
  1807→
  1808→      // Draw lines to nearby particles
  1809→      for (let j = i + 1; j < particles.length; j++) {
  1810→        const p2 = particles[j];
  1811→        const dx = p.x - p2.x;
  1812→        const dy = p.y - p2.y;
  1813→        const dist = Math.sqrt(dx * dx + dy * dy);
  1814→        if (dist < maxDistance) {
  1815→          ctx.beginPath();
  1816→          ctx.moveTo(p.x, p.y);
  1817→          ctx.lineTo(p2.x, p2.y);
  1818→          ctx.strokeStyle = `${color} / 0.06)`;
  1819→          ctx.lineWidth = 0.5;
  1820→          ctx.stroke();
  1821→        }
  1822→      }
  1823→    }
  1824→  }, [maxDistance, color]);
  1825→
  1826→  useEffect(() => {
  1827→    const canvas = canvasRef.current;
  1828→    if (!canvas) return;
  1829→    const ctx = canvas.getContext('2d');
  1830→    if (!ctx) return;
  1831→
  1832→    const w = window.innerWidth;
  1833→    const h = window.innerHeight;
  1834→    canvas.width = w;
  1835→    canvas.height = h;
  1836→    initParticles(w, h);
  1837→
  1838→    const animate = () => {
  1839→      draw(ctx, w, h);
  1840→      animRef.current = requestAnimationFrame(animate);
  1841→    };
  1842→    animate();
  1843→
  1844→    return () => cancelAnimationFrame(animRef.current);
  1845→  }, [initParticles, draw]);
  1846→
  1847→  // Handle resize
  1848→  useEffect(() => {

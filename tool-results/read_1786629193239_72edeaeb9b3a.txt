    60→function useInView(threshold = 0.15) {
    61→  const ref = useRef<HTMLDivElement>(null);
    62→  const [inView, setInView] = useState(false);
    63→  useEffect(() => {
    64→    const el = ref.current;
    65→    if (!el) return;
    66→    const obs = new IntersectionObserver(
    67→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    68→      { threshold }
    69→    );
    70→    obs.observe(el);
    71→    return () => obs.disconnect();
    72→  }, [threshold]);
    73→  return { ref, inView };
    74→}

  1230→function AnimatedCounter({ target, prefix = '', suffix = '' }: { target: number; prefix?: string; suffix?: string }) {
  1231→  const { ref, inView } = useInView(0.3);
  1232→  const [count, setCount] = useState(0);
  1233→
  1234→  useEffect(() => {
  1235→    if (!inView) return;
  1236→    let raf: number;
  1237→    const duration = 1200;
  1238→    const start = performance.now();
  1239→
  1240→    const step = (now: number) => {
  1241→      const elapsed = now - start;
  1242→      const progress = Math.min(elapsed / duration, 1);
  1243→      // Ease-out cubic
  1244→      const eased = 1 - Math.pow(1 - progress, 3);
  1245→      setCount(Math.round(eased * target));
  1246→      if (progress < 1) {
  1247→        raf = requestAnimationFrame(step);
  1248→      }
  1249→    };
  1250→    raf = requestAnimationFrame(step);
  1251→    return () => cancelAnimationFrame(raf);
  1252→  }, [inView, target]);
  1253→
  1254→  return (
  1255→    <span ref={ref}>
  1256→      {prefix}{count}{suffix}
  1257→    </span>
  1258→  );
  1259→}

    48→function useInView(threshold = 0.15) {
    49→  const ref = useRef<HTMLDivElement>(null);
    50→  const [inView, setInView] = useState(false);
    51→  useEffect(() => {
    52→    const el = ref.current;
    53→    if (!el) return;
    54→    const obs = new IntersectionObserver(
    55→      ([e]) => { if (e.isIntersecting) { setInView(true); obs.disconnect(); } },
    56→      { threshold }
    57→    );
    58→    obs.observe(el);
    59→    return () => obs.disconnect();
    60→  }, [threshold]);
    61→  return { ref, inView };
    62→}
